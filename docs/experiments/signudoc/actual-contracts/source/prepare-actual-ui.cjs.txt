const {request}=require('./upstream/node_modules/@playwright/test');
const {readFileSync,writeFileSync,readdirSync,mkdirSync}=require('node:fs');
const path=require('node:path');
const {mapSigningFields,assertLocalTrialUrl}=require('./prototype/field-map.cjs');
const origin=assertLocalTrialUrl('http://localhost:3187');
(async()=>{
 const sender=await request.newContext({baseURL:origin,extraHTTPHeaders:{origin}});
 const login=await sender.post('/api/auth/sign-in/email',{data:{email:'cmg-admin@example.com',password:'Synthetic-Only-Password-4130!x'}});
 if(login.status()!==200)throw Error('Synthetic login failed');
 const results=[];
 for(const variant of ['single','dual']){
  const bytes=readFileSync(path.join(__dirname,'actual-contract-evidence/input',variant+'-actual-cmg-prefilled.pdf'));
  const created=await sender.post('/api/envelopes',{multipart:{title:'ACTUAL CMG '+variant+' - SYNTHETIC UI DIAGNOSTIC',file:{name:variant+'.pdf',mimeType:'application/pdf',buffer:bytes}}});
  if(created.status()!==201)throw Error('Create failed '+await created.text());
  const {envelopeId}=await created.json();
  const names=variant==='single'?['Avery Synthetic']:['Avery Synthetic','Blake Synthetic'];
  const emails=names.map((_,i)=>'ui-'+variant+'-'+i+'@example.invalid');
  const parties=await sender.put('/api/envelopes/'+envelopeId+'/signers',{data:{signers:names.map((name,i)=>({name,email:emails[i],signOrder:i+1}))}});
  const partyList=(await parties.json()).signers;
  const ids=variant==='single'?{H1:partyList[0].id}:{H1:partyList[0].id,H2:partyList[1].id};
  const fields=await sender.put('/api/envelopes/'+envelopeId+'/fields',{data:{fields:mapSigningFields(variant,ids,{syntheticRequiredOnly:true})}});
  if(fields.status()!==200)throw Error('Fields failed');
  const sent=await sender.post('/api/envelopes/'+envelopeId+'/send',{data:{permittedUseConfirmed:true,expiresInDays:7,reminderIntervalDays:2}});
  if(sent.status()!==200)throw Error('Local capture send failed');
  let mail;
  for(let n=0;n<80;n++){
   mail=readdirSync(path.join(__dirname,'upstream/.data/e2e/outbox')).filter(x=>x.endsWith('.json')).map(x=>JSON.parse(readFileSync(path.join(__dirname,'upstream/.data/e2e/outbox',x)))).find(x=>x.template==='signing_invitation'&&x.to.includes(emails[0]));
   if(mail)break;
   await new Promise(r=>setTimeout(r,100));
  }
  if(!mail)throw Error('Capture missing');
  const url=(mail.text.match(/https?:\/\/[^\s<>"]+/g)||[]).find(u=>new URL(u).pathname.startsWith('/sign/'));
  results.push({variant,envelopeId,emails,signingUrl:url});
 }
 writeFileSync(path.join(__dirname,'actual-contract-evidence/ui-links-LOCAL-ONLY.json'),JSON.stringify(results,null,2),{mode:0o600});
 await sender.dispose();console.log('Prepared two synthetic-only actual-contract UI diagnostic envelopes.');
})().catch(e=>{console.error(e);process.exitCode=1});
