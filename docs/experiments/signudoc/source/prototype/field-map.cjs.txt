const assert = require("node:assert/strict");
const { readFileSync } = require("node:fs");
const path = require("node:path");

const manifest = JSON.parse(readFileSync(path.join(__dirname,"../cmg-source/lor-tag-manifest.json"), "utf8"));

/** Throwaway mapper for synthetic CMG-shaped PDFs. Not a production provider. */
function mapSigningFields(variant, signerIds, { syntheticRequiredOnly = false } = {}) {
  const definition = manifest.variants[variant];
  if (!definition) throw new Error("Unknown variant");
  const roles = variant === "single" ? ["H1"] : ["H1", "H2"];
  assert.deepEqual(Object.keys(signerIds).sort(), roles);
  if (new Set(Object.values(signerIds)).size !== roles.length) throw new Error("Distinct signers required");
  for (const role of roles) if (typeof signerIds[role] !== "string" || !signerIds[role].trim()) throw new Error("Missing signer");
  const optional = definition.signature.filter(f => f.optional);
  if (optional.length && !syntheticRequiredOnly) {
    throw new Error("BLOCKED: upstream requires all initials/signature/date fields; optional CMG authorization must remain optional");
  }
  return definition.signature.filter(f => !f.optional).map(f => {
    const type = ({ i: "initials", s: "signature", d: "date" })[f.type];
    if (!type || !roles.includes(f.role)) throw new Error("Unsupported field");
    const widthPt = type === "initials" ? 38 : type === "date" ? 74 : 145;
    const heightPt = 18;
    const field = {
      signerId: signerIds[f.role], type, page: f.page,
      x: f.x / 612, y: (792 - f.y - heightPt) / 792,
      width: widthPt / 612, height: heightPt / 792,
      required: true, label: f.id
    };
    if (field.x < 0 || field.y < 0 || field.x + field.width > 1 || field.y + field.height > 1) throw new Error("Out of bounds");
    return field;
  });
}

function assertLocalTrialUrl(value) {
  const u = new URL(value);
  if (u.protocol !== "http:" || !["127.0.0.1", "localhost"].includes(u.hostname) || u.port !== "3187") throw new Error("Only isolated trial origin allowed");
  return u.origin;
}

module.exports = {manifest,mapSigningFields,assertLocalTrialUrl};
